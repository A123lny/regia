# F1.2 — M2 Brand kit editabile (Implementation Plan)

> **Per agentic workers:** REQUIRED SUB-SKILL: `superpowers:subagent-driven-development` per eseguire task-per-task. Step usano `- [ ]` per tracciamento. Ogni task finisce con `git push origin feature/f1-2-brand-kit`.

**Goal:** rendere editabile per marchio il brand kit (tono di voce, contesto strategico, rubriche/pilastri, vincoli redazionali) fondamento per la generazione contenuti AI di M3.

**Architettura:** 4 nuove tabelle relazionali con FK cascade + RLS denial-by-default, seed iniziale dai contenuti del capitolato per i 4 marchi seed, nuovo modulo `m02-brand` con `public.ts` che espone getter + 11 server actions + helper `formatBrandContextForAI`, nuova route parametrica `/marchi/[slug]` con 7 card verticali (una per sezione: tono, contesto strategico, rubriche, parole bandite, argomenti sensibili, anteprima AI). Owner-only in F1.2.

**Tech stack:**
- shadcn/ui Card + Textarea (nuovo) + Input + Button + Table + Select + Badge
- Server Actions React 19 nativi (no client state library)
- Riuso adapter Supabase `getAdminClient` (RS-2) — service_role bypassa RLS
- Vitest per test invarianti DB (~4 nuovi file, ~28 test totali dopo F1.2)

- **RS-6** Convenzionalità: shadcn CLI, Server Actions React 19, `next/navigation`.
- **RS-8** Migrazioni naming `YYYYMMDDNNNNNN_verbo_soggetto.sql` (14 cifre continue).

// M02 Brand — repository: query DB + helper AI (RF-2.3, RF-2.3a, RF-2.4, RF-2.5).
// RS-2: usa getAdminClient() wrapper, no SDK Supabase diretto.

import { getAdminClient } from "@/lib/db/admin";
import type {
  BrandKit,
  BrandPillar,
  BrandForbiddenWord,
  BrandSensitiveTopic,
  BrandKitWithLists,
  SensitiveTopicKind,
} from "./brand-kit-types";

type BrandKitRow = {
  brand_id: string;
  tone_of_voice: string;
  strategic_purpose: string;
  strategic_year_goal: string;
  strategic_direction: string;
  updated_at: string;
};

type BrandPillarRow = {
  id: string;
  brand_id: string;
  name: string;
  weight: number;
  sort_order: number;
  created_at: string;
};

type BrandForbiddenWordRow = {
  id: string;
  brand_id: string;
  word: string;
  created_at: string;
};

type BrandSensitiveTopicRow = {
  id: string;
  brand_id: string;
  topic: string;
  kind: string;
  note: string;
  created_at: string;
};

function mapKit(row: BrandKitRow): BrandKit {
  return {
    brandId: row.brand_id,
    toneOfVoice: row.tone_of_voice,
    strategicPurpose: row.strategic_purpose,
    strategicYearGoal: row.strategic_year_goal,
    strategicDirection: row.strategic_direction,
    updatedAt: row.updated_at,
  };
}

function mapPillar(row: BrandPillarRow): BrandPillar {
  return {
    id: row.id,
    brandId: row.brand_id,
    name: row.name,
    weight: row.weight,
    sortOrder: row.sort_order,
    createdAt: row.created_at,
  };
}

function mapForbidden(row: BrandForbiddenWordRow): BrandForbiddenWord {
  return {
    id: row.id,
    brandId: row.brand_id,
    word: row.word,
    createdAt: row.created_at,
  };
}

function mapSensitive(row: BrandSensitiveTopicRow): BrandSensitiveTopic {
  return {
    id: row.id,
    brandId: row.brand_id,
    topic: row.topic,
    kind: row.kind as SensitiveTopicKind,
    note: row.note,
    createdAt: row.created_at,
  };
}

/**
 * Carica il brand kit completo di un marchio con tutte le liste correlate.
 * Ritorna null se il marchio non ha ancora un brand_kit (edge case: brand creato
 * dopo il seed senza upsert kit). In quel caso la UI crea la riga al primo save.
 */
export async function getBrandKitWithLists(
  brandId: string,
): Promise<BrandKitWithLists | null> {
  const admin = getAdminClient();

  const [kitRes, pillarsRes, forbRes, sensRes] = await Promise.all([
    admin.from("brand_kits").select("*").eq("brand_id", brandId).maybeSingle(),
    admin.from("brand_pillars").select("*").eq("brand_id", brandId).order("sort_order"),
    admin.from("brand_forbidden_words").select("*").eq("brand_id", brandId).order("word"),
    admin
      .from("brand_sensitive_topics")
      .select("*")
      .eq("brand_id", brandId)
      .order("kind")
      .order("topic"),
  ]);

  if (kitRes.error) throw new Error(`brand_kits query failed: ${kitRes.error.message}`);
  if (!kitRes.data) return null;

  return {
    kit: mapKit(kitRes.data as BrandKitRow),
    pillars: ((pillarsRes.data ?? []) as BrandPillarRow[]).map(mapPillar),
    forbiddenWords: ((forbRes.data ?? []) as BrandForbiddenWordRow[]).map(mapForbidden),
    sensitiveTopics: ((sensRes.data ?? []) as BrandSensitiveTopicRow[]).map(mapSensitive),
  };
}

/**
 * Formatta il brand kit come Markdown pronto per essere iniettato come
 * contesto obbligatorio nei prompt AI di M3 (RF-2.3, RF-2.3a).
 * Ritorna stringa vuota se il marchio non ha ancora un brand_kit configurato.
 */
export async function formatBrandContextForAI(brandId: string): Promise<string> {
  const bundle = await getBrandKitWithLists(brandId);
  if (!bundle) return "";

  const admin = getAdminClient();
  const { data: brandRow } = await admin
    .from("brands")
    .select("display_name")
    .eq("id", brandId)
    .single();
  const displayName = (brandRow?.display_name as string) ?? "Marchio";

  const lines: string[] = [];
  lines.push(`# Contesto di marca — ${displayName}`);
  lines.push("");
  lines.push("## Tono di voce");
  lines.push(bundle.kit.toneOfVoice || "_Non impostato._");
  lines.push("");
  lines.push("## Contesto strategico");
  lines.push(`- **Scopo e pubblico:** ${bundle.kit.strategicPurpose || "_Non impostato._"}`);
  lines.push(`- **Obiettivo dell'anno:** ${bundle.kit.strategicYearGoal || "_Non impostato._"}`);
  lines.push(`- **Direzione futura:** ${bundle.kit.strategicDirection || "_Non impostato._"}`);
  lines.push("");
  lines.push("## Rubriche (con peso di frequenza)");
  if (bundle.pillars.length === 0) {
    lines.push("_Nessuna rubrica configurata._");
  } else {
    for (const p of bundle.pillars) {
      lines.push(`- **${p.name}** (peso ${p.weight})`);
    }
  }
  lines.push("");
  lines.push("## Parole bandite (mai usare)");
  if (bundle.forbiddenWords.length === 0) {
    lines.push("_Nessuna parola bandita._");
  } else {
    lines.push(bundle.forbiddenWords.map((w) => w.word).join(", "));
  }
  lines.push("");
  lines.push("## Argomenti che richiedono attenzione");
  if (bundle.sensitiveTopics.length === 0) {
    lines.push("_Nessuno._");
  } else {
    for (const s of bundle.sensitiveTopics) {
      const kindLabel = s.kind === "claim_vietato" ? "claim vietato" : "revisione umana";
      const noteBit = s.note ? ` — ${s.note}` : "";
      lines.push(`- **${s.topic}** (${kindLabel})${noteBit}`);
    }
  }

  return lines.join("\n");
}

export async function addBrandPillar(
  brandId: string,
  brandSlug: string,
  payload: { name: string; weight: number },
): Promise<void> {
  const session = await requireOwner();
  if (payload.name.length < 1 || payload.name.length > 100) {
    throw new Error("Nome rubrica non valido (1-100 caratteri)");
  }
  if (payload.weight < 1 || payload.weight > 10) {

  const { data: maxRow } = await admin
    .from("brand_pillars")
    .select("sort_order")
    .eq("brand_id", brandId)
    .order("sort_order", { ascending: false })
    .limit(1)
    .maybeSingle();
  const nextSort = ((maxRow?.sort_order as number | undefined) ?? -1) + 1;

  const { error } = await admin.from("brand_pillars").insert({
    brand_id: brandId,
    name: payload.name,
    weight: payload.weight,
    sort_order: nextSort,
  });
  if (error) throw new Error(`Errore aggiunta rubrica: ${error.message}`);

  await logAuditEvent({
    organizationId: session.organizationId,
    actorUserId: session.userId,
    actorKind: "user",

}

export async function moveBrandPillar(
  brandId: string,
  brandSlug: string,
  pillarId: string,
  direction: "up" | "down",
): Promise<void> {
  const session = await requireOwner();
  const admin = getAdminClient();

  const { data: pillars, error: eList } = await admin
    .from("brand_pillars")
    .select("id, sort_order")
    .eq("brand_id", brandId)
    .order("sort_order");
  if (eList) throw new Error(`Errore lettura rubriche: ${eList.message}`);

  const list = (pillars ?? []) as Array<{ id: string; sort_order: number }>;
  const idx = list.findIndex((p) => p.id === pillarId);
  if (idx < 0) throw new Error("Rubrica non trovata");
  const swapIdx = direction === "up" ? idx - 1 : idx + 1;
  if (swapIdx < 0 || swapIdx >= list.length) return; // già in alto/basso, no-op

  const current = list[idx]!;
  const swap = list[swapIdx]!;
  // Scambia i sort_order via due update sequenziali (tabella piccola, no bulk).
  const { error: e1 } = await admin
    .from("brand_pillars")
    .update({ sort_order: swap.sort_order })
    .eq("id", current.id);
  if (e1) throw new Error(`Errore riordino (fase 1): ${e1.message}`);
  const { error: e2 } = await admin
    .from("brand_pillars")
    .update({ sort_order: current.sort_order })
    .eq("id", swap.id);
  if (e2) throw new Error(`Errore riordino (fase 2): ${e2.message}`);

  await logAuditEvent({
    organizationId: session.organizationId,
    actorUserId: session.userId,
    actorKind: "user",
    action: "brand_kit.pillar_moved",
    target: `brand:${brandSlug}`,
    details: { pillarId, direction },
  });
  revalidateBrandPage(brandSlug);
}

                      <div className="flex gap-1">
                        <form action={moveBrandPillar.bind(null, brandId, brandSlug, p.id, "up")}>

            action={async (formData: FormData) => {
              "use server";
              const word = String(formData.get("word") ?? "").trim();
              if (!word) return;
              await addBrandForbiddenWord(brandId, brandSlug, word);

            className="flex gap-2 border-t border-border pt-4"
          >
            <Input name="word" placeholder="Aggiungi parola" maxLength={100} required className="flex-1" />

            <Table>

                <TableRow>
                  <TableHead>Argomento</TableHead>
                  <TableHead className="w-40">Tipo</TableHead>
                  <TableHead>Nota</TableHead>
                  <TableHead className="w-10"></TableHead>

              </TableHeader>
              <TableBody>
                {sensitiveTopics.map((t) => (
                  <TableRow key={t.id}>
                    <TableCell>{t.topic}</TableCell>
                    <TableCell>
                      {t.kind === "claim_vietato" ? (
                        <Badge className="border border-danger/30 bg-danger/15 text-danger">
                          Claim vietato
                        </Badge>
                      ) : (
                        <Badge className="border border-warning/30 bg-warning/15 text-warning">
                          Revisione umana
                        </Badge>
                      )}
                    </TableCell>
                    <TableCell className="text-text-muted">{t.note || "—"}</TableCell>
                    <TableCell>
                      <form
                        action={removeBrandSensitiveTopic.bind(null, brandId, brandSlug, t.id)}
                      >
                        <Button
                          type="submit"
                          variant="ghost"
                          size="icon"
                          aria-label="Rimuovi"
                          className="text-danger hover:bg-danger/10"
                        >
                          <Trash2 className="h-4 w-4" />
                        </Button>
                      </form>
                    </TableCell>
                  </TableRow>
                ))}
              </TableBody>
            </Table>
          )}

          <form
            action={async (formData: FormData) => {
              "use server";
              const topic = String(formData.get("topic") ?? "");
              const kind = String(formData.get("kind") ?? "revisione_umana") as
                | "claim_vietato"
                | "revisione_umana";
              const note = String(formData.get("note") ?? "");
              await addBrandSensitiveTopic(brandId, brandSlug, { topic, kind, note });
            }}
            className="flex flex-col gap-2 border-t border-border pt-4 sm:flex-row sm:items-end"
          >
            <div className="flex-1">
              <Label htmlFor="new-topic">Nuovo argomento</Label>
              <Input id="new-topic" name="topic" required maxLength={200} placeholder="es. Sconti carburante" />
            </div>
            <div className="w-40">
              <Label htmlFor="new-topic-kind">Tipo</Label>
              <Select name="kind" defaultValue="revisione_umana">
                <SelectTrigger id="new-topic-kind">
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  <SelectItem value="revisione_umana">Revisione umana</SelectItem>
                  <SelectItem value="claim_vietato">Claim vietato</SelectItem>
                </SelectContent>
              </Select>
            </div>
            <div className="flex-1">
              <Label htmlFor="new-topic-note">Nota (opzionale)</Label>
              <Input id="new-topic-note" name="note" maxLength={500} placeholder="Motivo o riferimento" />
            </div>
            <Button type="submit" className="bg-primary text-white hover:bg-primary-hover">
              Aggiungi
            </Button>
          </form>
        </CardContent>
      </Card>

      {/* Card 6: Anteprima contesto AI */}
      <Card>
        <CardHeader>
          <CardTitle className="font-serif">Anteprima contesto per AI</CardTitle>
          <CardDescription>
            Come l'AI di M3 leggerà il brand kit di questo marchio quando genererà
            contenuti (RF-2.3, RF-2.3a). Aggiornato in tempo reale.
          </CardDescription>
        </CardHeader>
        <CardContent>
          <pre className="max-h-96 overflow-auto rounded-md border border-border bg-surface-muted p-4 text-xs whitespace-pre-wrap">
            {aiContext || "_Nessun brand kit configurato._"}
          </pre>
        </CardContent>
      </Card>
    </div>
  );
